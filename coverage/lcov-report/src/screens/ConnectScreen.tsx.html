/**

// HUD label positioning inside the radar box.
function hud(corner: 'tl' | 'tr' | 'bl' | 'br', color: string = colors.muted) {
  const base = { position: 'absolute' as const, fontSize: 10, fontWeight: '700' as const, letterSpacing: 1.2, color };
  const pos = {
    tl: { top: 4, left: 2 },
    tr: { top: 4, right: 2 },
    bl: { bottom: 4, left: 2 },
    br: { bottom: 4, right: 2 },
  }[corner];
  return { ...base, ...pos };
}
